<script>
  let timerStarted = false;
  let timeLeft = 60;
  let timerInterval;
  let currentIndex = 0;
  let question = [
    "What is the default value of a boolean variable in Java?",
    "Which of the following is not a valid data type in Java?",
    "What is the purpose of the 'final' keyword in Java?",
    "Which class is the superclass of all Java classes?",
    "Which method is used to start a thread in Java?",
    "Which of the following is a wrapper class in Java?",
    "Which method in Java is used to print output to the console?",
    "Which of the following can be used to create an immutable object in Java?",
    "In Java, which statement is used to exit from a loop?",
    "What does JVM stand for?"
  ];
  let opt1 = ["false", "float", "To define constants", "Object", "start()", "Integer", "print()", "By using the 'final' keyword", "continue", "Java Virtual Machine"];
  let opt2 = ["true", "double", "To prevent inheritance", "Class", "run()", "String", "System.out.println()", "By overriding the 'toString()' method", "break", "Java Verified Machine"];
  let opt3 = ["null", "char", "To prevent changes in the variable's value", "Throwable", "execute()", "Long", "println()", "By using the 'private' modifier", "exit", "Java Virtual Macro"];
  let opt4 = ["0", "boolean", "To mark methods as abstract", "Serializable", "init()", "Byte", "printIn()", "By using the 'volatile' keyword", "stop", "Java Virtual Machine"];
  let correctans = ["false", "float", "To define constants", "Object", "start()", "Integer", "System.out.println()", "By using the 'final' keyword", "break", "Java Virtual Machine"];
  let score = new Array(10).fill(0);

  function showQuestion(index) {
    if (!timerStarted) {
      startTimer();
      timerStarted = true;
      document.getElementById("control-buttons").classList.remove("hidden"); // Show buttons when quiz starts
    }
    currentIndex = index;
    document.getElementById("question-text").innerHTML = question[index];
    document.getElementById("options").innerHTML = `
      <input type='radio' name='x' value='${opt1[index]}'> ${opt1[index]}<br>
      <input type='radio' name='x' value='${opt2[index]}'> ${opt2[index]}<br>
      <input type='radio' name='x' value='${opt3[index]}'> ${opt3[index]}<br>
      <input type='radio' name='x' value='${opt4[index]}'> ${opt4[index]}
    `;
    responsiveVoice.speak(question[index], "UK English Male"); // Speak question
    responsiveVoice.speak("Option 1: " + opt1[index], "UK English Male"); // Speak options
    responsiveVoice.speak("Option 2: " + opt2[index], "UK English Male");
    responsiveVoice.speak("Option 3: " + opt3[index], "UK English Male");
    responsiveVoice.speak("Option 4: " + opt4[index], "UK English Male");
  }

  function startTimer() {
    timerInterval = setInterval(function() {
      if (timeLeft <= 0) {
        clearInterval(timerInterval);
        submitQuiz();
      } else {
        timeLeft--;
        let minutes = Math.floor(timeLeft / 60);
        let seconds = timeLeft % 60;
        document.getElementById("timer").textContent = `Time Left: ${minutes}:${seconds < 10 ? '0' + seconds : seconds}`;
      }
    }, 1000);
  }

  function showAnswer() {
    const selectedOption = document.querySelector('input[name="x"]:checked');
    if (selectedOption) {
      const answer = selectedOption.value;
      if (correctans[currentIndex] === answer) {
        score[currentIndex] = 1;
        alert("Correct!");
        document.getElementById("question-list").children[currentIndex].style.backgroundColor = "#27ae60"; // Green for correct
      } else {
        alert("Incorrect!");
        document.getElementById("question-list").children[currentIndex].style.backgroundColor = "#e74c3c"; // Red for incorrect
      }
    } else {
      alert("Please select an answer before checking.");
    }
  }

  function next() {
    if (currentIndex < 9) {
      showQuestion(++currentIndex);
    }
  }

  function previous() {
    if (currentIndex > 0) {
      showQuestion(--currentIndex);
    }
  }

  function submitQuiz() {
    clearInterval(timerInterval);
    let totalScore = score.reduce((a, b) => a + b, 0);
    alert("Quiz Completed! Your score: " + totalScore + "/10");
  }

  function reListen() {
    responsiveVoice.speak(question[currentIndex], "UK English Male"); // Re-listen to question
  }
</script>
